    return out

def vig(text,key,enc=True):
    out=''; key=key.upper()
    for idx,ch in enumerate(text.upper()):
        x=ord(ch)-65; k=ord(key[idx%len(key)])-65
        y=(x+(k if enc else -k))%26; out+=chr(65+y)
    return out

def to_base(n,b):
    digits='0123456789ABCDEF'; s=''
    if n==0:return '0'
    while n: s=digits[n%b]+s; n//=b
    return s
