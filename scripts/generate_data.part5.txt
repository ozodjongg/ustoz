    return opts, opts.index(str(correct))

def qrec(slug,i,diff,prompt,correct,distractors,explanation):
    opts,ans=shuffle_mcq(correct,distractors,R)
    pts={'easy':0.9,'medium':1.5,'hard':2.6}[diff]
    return {'id':f'{slug}-{i:03d}','unit':slug,'difficulty':diff,'points':pts,'prompt':prompt,'options':opts,'answer':ans,'explanation':explanation}

def caesar(text,k,enc=True):
    out=''
    for ch in text:
        if ch.isalpha():
            x=ord(ch)-65; y=(x+(k if enc else -k))%26; out+=chr(65+y)
        else: out+=ch
