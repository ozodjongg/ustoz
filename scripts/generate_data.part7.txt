def gen_multimedia(i,d):
    typ=i%4
    if typ==0:
        sec=R.choice([30,60,90,120,150]); fps=R.choice([24,25,30,60]); c=sec*fps
        return qrec('multimedia',i,d,f"Video {sec} sekund, {fps} FPS. Jami nechta kadr?",c,[sec+fps,c-fps,c+fps],"Kadrlar soni=vaqt×FPS.")
    if typ==1:
        sec=R.choice([10,20,30,60]); hz=R.choice([8000,16000,22000]); bits=R.choice([8,16]); ch=R.choice([1,2]); bytes_=sec*hz*bits*ch//8
        kb=bytes_//1024
        return qrec('multimedia',i,d,f"{sec}s, {hz}Hz, {bits}bit, {ch} kanal audio xom hajmi taxminan necha KB (1024)?",kb,[max(1,kb//2),kb*2,bytes_],"sekund×Hz×bit×kanal /8 /1024.")
    if typ==2:
        return qrec('multimedia',i,d,"Qaysi kengaytma odatda video faylga tegishli?",".mkv",['.wav','.flac','.aac'],".mkv video konteyner formati.")
    w,h=R.choice([(320,240),(640,480),(800,600)]); depth=R.choice([8,24]); bytes_=w*h*depth//8
    return qrec('multimedia',i,d,f"{w}×{h} piksel, {depth} bit rang chuqurligidagi bitta xom kadr necha byte?",bytes_,[w*h,bytes_//8,bytes_*8],"piksel soni×bit/piksel /8.")

def gen_encoding(i,d):
    typ=i%3

    if typ==1:
        ch=R.choice(list('ABCabc012')); code=ord(ch)
        return qrec('encoding-bits',i,d,f"ASCII bo‘yicha '{ch}' belgining o‘nlik kodi?",code,[code-1,code+1,code+32],"ASCII jadvalida har belgining sonli kodi bor.")
    chars=R.choice([100,256,512,1024]); c=chars
    return qrec('encoding-bits',i,d,f"1 belgi=1 byte deb olinsa, {chars} belgili matn hajmi necha byte?",c,[chars*8,chars//8,chars+1024],"Har belgi 1 byte bo‘lsa, byte soni belgilar soniga teng.")
