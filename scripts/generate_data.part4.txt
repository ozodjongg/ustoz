def shuffle_mcq(correct, distractors, rng):
    opts=[str(correct)]+[str(x) for x in distractors]
    # unique preserving
    uniq=[]
    for x in opts:
        if x not in uniq: uniq.append(x)
    while len(uniq)<4:
        candidate=str(rng.randint(1,999))
        if candidate not in uniq: uniq.append(candidate)
    opts=uniq[:4]
    rng.shuffle(opts)
